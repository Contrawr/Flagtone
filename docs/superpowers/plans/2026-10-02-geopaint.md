# GeoPaint Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A browser game in which the player recreates a flag's one hidden color with hue/saturation/brightness sliders and is scored 0–10 by CIEDE2000 distance. It has two modes: Standard (5 flags, out of 50) and Infinite (3 lives, a life lost below 7.0). There are 4 difficulty tiers by flag obscurity.

**Architecture:** A static site with plain ES modules and no build step. Pure logic (`colors.js`, `flags.js`, `game.js`, plus the pixel-tally helpers in `flagLoader.js`) is unit-tested with `node --test`. DOM code (`flagLoader.js` fetch/raster, `ui.js`) is verified in the in-app browser. Flag SVGs come from the pinned `flag-icons@7.5.0` package on jsDelivr.

**Tech Stack:** HTML, CSS and vanilla JS (ES modules). Node 26 built-in test runner. `python3 -m http.server` for local serving. Google Fonts (Figtree).

**Spec:** `docs/superpowers/specs/2026-10-02-geopaint-design.md`

All code in this plan was prototyped and run before the plan was written: 22/22 unit tests pass, all 197 flags pass the check page, and both modes were played through in the browser at 375 px width.

## Global Constraints

- No build step, no npm dependencies. `package.json` exists only for `"type": "module"` and the `test` script.
- Flag URL: `https://cdn.jsdelivr.net/npm/flag-icons@7.5.0/flags/4x3/{code}.svg` (version pinned).
- Score: `clamp(10 − ΔE2000 / 4, 0, 10)`, rounded to 1 decimal. Infinite loses a life when `score < 7.0` (exactly 7.0 is safe).
- Standard = 5 rounds, no repeats. Infinite = 3 lives, reshuffle on exhaustion, no back-to-back repeat.
- A color is hideable only if it covers ≥ 8% of opaque pixels (RGB tolerance 12, 160×120 raster).
- Up to 5 load attempts per round, then an error with Retry.
- Best scores in `localStorage` under `geopaint:best:{mode}:{tier}`; all storage access wrapped in try/catch.
- Layout: max content width 520px, 16px side gutter, no horizontal scroll at 375px. Light theme only.
- Run tests with `npm test` (= `node --test`). Note that `node --test test/` with a directory argument fails on Node 26, so don't use that form.

## Deviations from the spec (intentional)

- Prefetching lives in `ui.js` (a `pending` promise), not as a cache inside `flagLoader.js`. The loader stays stateless.
- `game.submit(code, guessHex, trueHex)` takes the flag code too, so the result can carry the country name.
- `flagLoader.js` also exports the pure helpers `tallyColors` and `pickCandidates` for unit testing, plus `analyzeFlag` for the dev check page.

```js
// Pure color math: HSB/RGB/hex conversions, CIELAB, CIEDE2000, scoring.

export function clamp(x, lo, hi) {
  return Math.min(hi, Math.max(lo, x));
}

// h in degrees [0, 360], s and v in [0, 1]. Returns integer channels 0–255.
export function hsbToRgb(h, s, v) {
  const f = (n) => {
    const k = (n + h / 60) % 6;
    return v - v * s * Math.max(0, Math.min(k, 4 - k, 1));
  };
  return { r: Math.round(f(5) * 255), g: Math.round(f(3) * 255), b: Math.round(f(1) * 255) };
}

export function rgbToHex({ r, g, b }) {
  return '#' + [r, g, b].map((c) => c.toString(16).padStart(2, '0')).join('');
}

export function hexToRgb(hex) {
  const n = parseInt(hex.slice(1), 16);
  return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 };
}

export function hsbToHex(h, s, v) {
  return rgbToHex(hsbToRgb(h, s, v));
}

const NAMED = {
  white: '#ffffff', black: '#000000', red: '#ff0000', green: '#008000',
  blue: '#0000ff', yellow: '#ffff00', gold: '#ffd700', gray: '#808080',
  grey: '#808080', silver: '#c0c0c0', navy: '#000080', maroon: '#800000',
  orange: '#ffa500', purple: '#800080',
};

// Any SVG fill value → '#rrggbb', or null when it is not a plain color.
export function normalizeColor(value) {
  if (value == null) return null;
  const v = String(value).trim().toLowerCase();
  if (/^#[0-9a-f]{6}$/.test(v)) return v;
  if (/^#[0-9a-f]{3}$/.test(v)) return '#' + [...v.slice(1)].map((c) => c + c).join('');
  const rgb = v.match(/^rgb\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\s*\)$/);
  if (rgb) return rgbToHex({ r: +rgb[1], g: +rgb[2], b: +rgb[3] });
  return NAMED[v] ?? null;
}

// sRGB (0–255) → CIELAB, D65 white point.
export function rgbToLab({ r, g, b }) {
  const lin = (c) => {
    c /= 255;
    return c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  };
  const R = lin(r), G = lin(g), B = lin(b);
  const X = (0.4124564 * R + 0.3575761 * G + 0.1804375 * B) / 0.95047;
  const Y = 0.2126729 * R + 0.7151522 * G + 0.072175 * B;
  const Z = (0.0193339 * R + 0.119192 * G + 0.9503041 * B) / 1.08883;
  const f = (t) => (t > 216 / 24389 ? Math.cbrt(t) : (24389 / 27 * t + 16) / 116);
  const fx = f(X), fy = f(Y), fz = f(Z);
  return { L: 116 * fy - 16, a: 500 * (fx - fy), b: 200 * (fy - fz) };
}

// CIEDE2000 color difference (Sharma, Wu & Dalal 2005), kL = kC = kH = 1.
export function deltaE2000(c1, c2) {
  const rad = Math.PI / 180;
  const { L: L1, a: a1, b: b1 } = c1;
  const { L: L2, a: a2, b: b2 } = c2;
  const Cbar = (Math.hypot(a1, b1) + Math.hypot(a2, b2)) / 2;
  const G = 0.5 * (1 - Math.sqrt(Cbar ** 7 / (Cbar ** 7 + 25 ** 7)));
  const a1p = (1 + G) * a1;
  const a2p = (1 + G) * a2;
  const C1p = Math.hypot(a1p, b1);
  const C2p = Math.hypot(a2p, b2);
  const hue = (b, ap) => {
    if (b === 0 && ap === 0) return 0;
    const h = Math.atan2(b, ap) / rad;
    return h >= 0 ? h : h + 360;
  };
  const h1p = hue(b1, a1p);
  const h2p = hue(b2, a2p);

  const dLp = L2 - L1;
  const dCp = C2p - C1p;
  let dhp = 0;
  if (C1p * C2p !== 0) {
    dhp = h2p - h1p;
    if (dhp > 180) dhp -= 360;
    else if (dhp < -180) dhp += 360;
  }
  const dHp = 2 * Math.sqrt(C1p * C2p) * Math.sin((dhp / 2) * rad);

  const Lbarp = (L1 + L2) / 2;
  const Cbarp = (C1p + C2p) / 2;
  let hbarp = h1p + h2p;
  if (C1p * C2p !== 0) {
    if (Math.abs(h1p - h2p) <= 180) hbarp = (h1p + h2p) / 2;
    else if (h1p + h2p < 360) hbarp = (h1p + h2p + 360) / 2;
    else hbarp = (h1p + h2p - 360) / 2;
  }
  const T = 1
    - 0.17 * Math.cos((hbarp - 30) * rad)
    + 0.24 * Math.cos(2 * hbarp * rad)
    + 0.32 * Math.cos((3 * hbarp + 6) * rad)
    - 0.2 * Math.cos((4 * hbarp - 63) * rad);
  const dTheta = 30 * Math.exp(-(((hbarp - 275) / 25) ** 2));
  const Rc = 2 * Math.sqrt(Cbarp ** 7 / (Cbarp ** 7 + 25 ** 7));
  const Sl = 1 + (0.015 * (Lbarp - 50) ** 2) / Math.sqrt(20 + (Lbarp - 50) ** 2);
  const Sc = 1 + 0.045 * Cbarp;
  const Sh = 1 + 0.015 * Cbarp * T;
  const Rt = -Math.sin(2 * dTheta * rad) * Rc;
  const l = dLp / Sl, c = dCp / Sc, h = dHp / Sh;
  return Math.sqrt(l * l + c * c + h * h + Rt * c * h);
}

// ΔE → 0–10, one decimal. ΔE 0 → 10, ΔE 12 → 7, ΔE ≥ 40 → 0.
export function scoreFromDeltaE(dE) {
  return Math.round(clamp(10 - dE / 4, 0, 10) * 10) / 10;
}

export function matchScore(guessHex, trueHex) {
  const deltaE = deltaE2000(rgbToLab(hexToRgb(guessHex)), rgbToLab(hexToRgb(trueHex)));
  return { deltaE, score: scoreFromDeltaE(deltaE) };
}
```

```js
// Game state machine. No DOM access: storage, randomness and scoring are injected.
import { matchScore } from './colors.js';
import { poolFor, flagByCode } from './flags.js';

export const STANDARD_ROUNDS = 5;
export const START_LIVES = 3;
export const LIFE_THRESHOLD = 7;

function shuffle(items, random) {
  const a = [...items];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

export class Game {
  constructor({ storage = null, random = Math.random, score = matchScore } = {}) {
    this.storage = storage;
    this.random = random;
    this.score = score;
    this.mode = null;
    this.tier = null;
  }

  start(mode, tier) {
    this.mode = mode;
    this.tier = tier;
    this.results = [];
    this.lives = START_LIVES;
    this.lastCode = null;
    this.queue = [];
    this.isNewBest = false;
  }

  // Next flag to show. Standard never repeats; Infinite reshuffles the pool when
  // it runs out and never shows the same flag twice in a row.
  nextFlagCode() {
    if (this.queue.length === 0) {
      const codes = poolFor(this.tier).map((f) => f.code);
      this.queue = shuffle(codes, this.random);
      if (this.queue[0] === this.lastCode && this.queue.length > 1) {
        [this.queue[0], this.queue[1]] = [this.queue[1], this.queue[0]];
      }
    }
    this.lastCode = this.queue.shift();
    return this.lastCode;
  }

  get round() {
    return this.results.length + 1;
  }

  get total() {
    return Math.round(this.results.reduce((sum, r) => sum + r.score, 0) * 10) / 10;
  }

  submit(code, guessHex, trueHex) {
    const { deltaE, score } = this.score(guessHex, trueHex);
    const lostLife = this.mode === 'infinite' && score < LIFE_THRESHOLD;
    if (lostLife) this.lives -= 1;
    const result = { code, name: flagByCode(code).name, guessHex, trueHex, deltaE, score, lostLife };
    this.results.push(result);
    if (this.isOver()) this.saveBest();
    return result;
  }

  isOver() {
    if (this.mode === 'standard') return this.results.length >= STANDARD_ROUNDS;
    return this.lives <= 0;
  }

  // Standard: total out of 50. Infinite: flags completed (the streak).
  finalScore() {
    return this.mode === 'standard' ? this.total : this.results.length;
  }

  bestFor(mode, tier) {
    try {
      const raw = this.storage?.getItem(`geopaint:best:${mode}:${tier}`);
      return raw == null ? null : Number(raw);
    } catch {
      return null;
    }
  }

  saveBest() {
    const previous = this.bestFor(this.mode, this.tier);
    const current = this.finalScore();
    if (previous == null || current > previous) {
      this.isNewBest = true;
      try {
        this.storage?.setItem(`geopaint:best:${this.mode}:${this.tier}`, String(current));
      } catch {
        // Storage unavailable (private mode): the best score just isn't kept.
      }
    }
  }
}
```

How it works:
1. `analyzeFlag` fetches the SVG and parses it with `DOMParser`.
2. Every shape with no fill anywhere up its tree gets an explicit `#000000`, so a hidden black also repaints implicitly-black shapes.
3. It collects every fill, normalized to `#rrggbb`. flag-icons uses hex plus the named colors red, gold, gray and green; there are no `<style>` blocks.
4. It rasterizes the SVG to a 160×120 canvas and keeps fills covering ≥ 8% of the area. If the canvas can't be read, it falls back to the fills of top-level shapes.
5. `loadFlag` picks one candidate at random. It then sets `style.fill` on every element whose own fill equals that color; inline style beats the `fill` attribute, and children inherit from a recolored `<g>`.

```js
// Fetches a flag SVG, finds which colors cover enough area to be hidden,
// and returns an inline SVG whose hidden color can be repainted live.
import { normalizeColor, hexToRgb } from './colors.js';

export const flagUrl = (code) =>
  `https://cdn.jsdelivr.net/npm/flag-icons@7.5.0/flags/4x3/${code}.svg`;

export const MIN_SHARE = 0.08;
const MATCH_TOLERANCE = 12;
const RASTER_W = 160;
const RASTER_H = 120;
const SHAPES = new Set(['path', 'rect', 'circle', 'ellipse', 'polygon', 'polyline', 'text']);
const NON_RENDERED = 'defs, clipPath, mask, symbol, pattern, marker';

// Pure: counts opaque pixels per palette color. A pixel counts toward the nearest
// palette color if it lies within MATCH_TOLERANCE (RGB distance); anti-aliased
// edge pixels usually don't.
export function tallyColors(data, palette, tolerance = MATCH_TOLERANCE) {
  const rgbs = palette.map(hexToRgb);
  const counts = new Map(palette.map((hex) => [hex, 0]));
  const tol2 = tolerance * tolerance;
  let total = 0;
  for (let i = 0; i < data.length; i += 4) {
    if (data[i + 3] < 128) continue;
    total++;
    let best = -1;
    let bestD = Infinity;
    for (let p = 0; p < rgbs.length; p++) {
      const dr = data[i] - rgbs[p].r, dg = data[i + 1] - rgbs[p].g, db = data[i + 2] - rgbs[p].b;
      const d = dr * dr + dg * dg + db * db;
      if (d < bestD) { bestD = d; best = p; }
    }
    if (best >= 0 && bestD <= tol2) counts.set(palette[best], counts.get(palette[best]) + 1);
  }
  return { counts, total };
}

// Pure: colors covering at least minShare of the opaque pixels.
export function pickCandidates(counts, total, minShare = MIN_SHARE) {
  if (total === 0) return [];
  return [...counts].filter(([, n]) => n / total >= minShare).map(([hex]) => hex);
}

function fillOf(el) {
  return normalizeColor(el.getAttribute('fill') ?? el.style?.fill ?? null);
}

function parseSvg(text) {
  const doc = new DOMParser().parseFromString(text, 'image/svg+xml');
  if (doc.querySelector('parsererror')) throw new Error('SVG parse error');
  return document.importNode(doc.documentElement, true);
}

// Shapes with no fill anywhere up the tree render black by default. Make that
// explicit so a hidden black also repaints them.
function makeDefaultFillsExplicit(svg) {
  for (const el of svg.querySelectorAll('*')) {
    if (!SHAPES.has(el.localName) || el.closest(NON_RENDERED)) continue;
    let node = el;
    let hasFill = false;
    while (node && node !== svg.parentNode) {
      if (node.getAttribute?.('fill') || node.style?.fill) { hasFill = true; break; }
      node = node.parentNode;
    }
    if (!hasFill) el.setAttribute('fill', '#000000');
  }
}

function collectFills(svg) {
  const set = new Set();
  for (const el of svg.querySelectorAll('*')) {
    const hex = fillOf(el);
    if (hex) set.add(hex);
  }
  return [...set];
}

// Fallback when the canvas can't be read: fills of top-level shapes (usually stripes).
function topLevelFills(svg) {
  const set = new Set();
  for (const el of svg.children) {
    const hex = fillOf(el);
    if (hex && el.localName !== 'defs') set.add(hex);
  }
  return [...set];
}

function rasterize(svg) {
  const copy = svg.cloneNode(true);
  copy.setAttribute('width', RASTER_W);
  copy.setAttribute('height', RASTER_H);
  const text = new XMLSerializer().serializeToString(copy);
  const url = URL.createObjectURL(new Blob([text], { type: 'image/svg+xml' }));
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => {
      try {
        const canvas = document.createElement('canvas');
        canvas.width = RASTER_W;
        canvas.height = RASTER_H;
        const ctx = canvas.getContext('2d', { willReadFrequently: true });
        ctx.drawImage(img, 0, 0, RASTER_W, RASTER_H);
        resolve(ctx.getImageData(0, 0, RASTER_W, RASTER_H).data);
      } catch (err) {
        reject(err);
      } finally {
        URL.revokeObjectURL(url);
      }
    };
    img.onerror = () => {
      URL.revokeObjectURL(url);
      reject(new Error('SVG rasterize failed'));
    };
    img.src = url;
  });
}

// Fetch + parse + find hideable colors. Used by loadFlag and the dev check page.
export async function analyzeFlag(code) {
  const res = await fetch(flagUrl(code));
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  const svg = parseSvg(await res.text());
  makeDefaultFillsExplicit(svg);
  const palette = collectFills(svg);
  let candidates;
  try {
    const { counts, total } = tallyColors(await rasterize(svg), palette);
    candidates = pickCandidates(counts, total);
  } catch {
    candidates = topLevelFills(svg);
  }
  return { svg, candidates };
}

export async function loadFlag(code, random = Math.random) {
  const { svg, candidates } = await analyzeFlag(code);
  if (candidates.length === 0) throw new Error('no color covers enough area');
  const hiddenHex = candidates[Math.floor(random() * candidates.length)];
  const hidden = [...svg.querySelectorAll('*')].filter((el) => fillOf(el) === hiddenHex);
  svg.removeAttribute('width');
  svg.removeAttribute('height');
  svg.setAttribute('class', 'flag-svg');
  svg.setAttribute('role', 'img');
  return {
    svg,
    hiddenHex,
    setGuess(hex) {
      for (const el of hidden) el.style.fill = hex;
    },
    reveal() {
      for (const el of hidden) el.style.removeProperty('fill');
    },
  };
}
```

```js
// DOM rendering and event wiring. Game rules live in game.js; this file only draws.
import { Game, STANDARD_ROUNDS, START_LIVES } from './game.js';
import { TIERS, flagByCode } from './flags.js';
import { loadFlag } from './flagLoader.js';
import { hsbToHex } from './colors.js';

const TIER_LABELS = { easy: 'Easy', medium: 'Medium', hard: 'Hard', impossible: 'Impossible' };
const MODE_LABELS = { standard: 'Standard', infinite: 'Infinite' };
const MAX_LOAD_ATTEMPTS = 5;
const HUE_GRADIENT =
  'linear-gradient(to right, #ff0000, #ffff00, #00ff00, #00ffff, #0000ff, #ff00ff, #ff0000)';

const LOGO = `<svg class="logo" viewBox="0 0 48 48" aria-hidden="true">
  <g transform="rotate(-8 22 26)">
    <rect x="4" y="11" width="36" height="30" rx="7" fill="#fff" stroke="#ddd6cf" stroke-width="2"/>
    <rect x="10" y="17" width="8" height="18" fill="#3b6fd4"/>
    <rect x="18" y="17" width="8" height="18" fill="#f5f1ed"/>
    <rect x="26" y="17" width="8" height="18" fill="#e04848"/>
  </g>
  <path d="M33 14 41 4l4 3-8 11z" fill="#2b2a29"/>
  <path d="M31 15c2-1 4 0 5 2-1 3-4 4-7 3 1-2 0-4 2-5z" fill="#2b2a29"/>
</svg>`;
const HOME_ICON = `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="M15 18 9 12l6-6"/></svg>`;
const CHECK_ICON = `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.6" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="m5 13 4 4L19 7"/></svg>`;

function safeStorage() {
  try {
    const s = window.localStorage;
    s.getItem('geopaint:probe');
    return s;
  } catch {
    return null;
  }
}

function verdictFor(score) {
  if (score >= 9.5) return 'Perfect match!';
  if (score >= 8.5) return 'Excellent';
  if (score >= 7) return 'Close';
  if (score >= 4) return 'Not quite';
  return 'Way off';
}

function formatBest(mode, best) {
  if (best == null) return 'No best score yet';
  return mode === 'standard' ? `Best: ${best} / 50` : `Best streak: ${best}`;
}

export function boot(app) {
  const game = new Game({ storage: safeStorage() });
  const choice = { mode: 'standard', tier: 'easy' };
  let pending = null;

  function header({ sub, left = '', right = '' }) {
    return `<header class="header">
      <div class="header-left">${left}</div>
      <div class="brand">${LOGO}<div><div class="brand-name">GeoPaint</div><div class="brand-sub">${sub}</div></div></div>
      <div class="header-right">${right}</div>
    </header>`;
  }

  function gameHeader() {
    const right = game.mode === 'standard'
      ? `<span class="pill">${Math.min(game.round, STANDARD_ROUNDS)} / ${STANDARD_ROUNDS}</span>`
      : `<span class="hearts" aria-label="${game.lives} ${game.lives === 1 ? 'life' : 'lives'} left">${'♥'.repeat(game.lives)}<span class="lost">${'♥'.repeat(START_LIVES - game.lives)}</span></span>
         <span class="pill" aria-label="Streak">${game.results.length}</span>`;
    return header({
      sub: `${MODE_LABELS[game.mode]} · ${TIER_LABELS[game.tier]}`,
      left: `<button class="icon-btn" id="quit" aria-label="Back to home">${HOME_ICON}</button>`,
      right,
    });
  }

  function wireQuit() {
    app.querySelector('#quit').addEventListener('click', renderHome);
  }

  // ---------- Home ----------
  function renderHome() {
    pending = null;
    app.innerHTML = `${header({ sub: 'Flag color matching' })}
      <section>
        <h1 class="title">Match the missing color</h1>
        <p class="lead">Every flag is missing one color.<br>Mix it back in with the sliders.</p>
        <div class="field">
          <span class="label">Mode</span>
          <div class="seg" data-group="mode">
            <button data-value="standard">Standard<small>5 flags</small></button>
            <button data-value="infinite">Infinite<small>3 lives</small></button>
          </div>
        </div>
        <div class="field">
          <span class="label">Difficulty</span>
          <div class="seg seg-4" data-group="tier">
            ${TIERS.map((t) => `<button data-value="${t}">${TIER_LABELS[t]}</button>`).join('')}
          </div>
        </div>
        <p class="best" id="best"></p>
        <button class="primary" id="start">Start</button>
      </section>`;

    const sync = () => {
      for (const group of app.querySelectorAll('.seg')) {
        for (const btn of group.querySelectorAll('button')) {
          btn.setAttribute('aria-pressed', String(choice[group.dataset.group] === btn.dataset.value));
        }
      }
      app.querySelector('#best').textContent = formatBest(choice.mode, game.bestFor(choice.mode, choice.tier));
    };
    for (const group of app.querySelectorAll('.seg')) {
      group.addEventListener('click', (e) => {
        const btn = e.target.closest('button');
        if (!btn) return;
        choice[group.dataset.group] = btn.dataset.value;
        sync();
      });
    }
    app.querySelector('#start').addEventListener('click', () => {
      game.start(choice.mode, choice.tier);
      pending = null;
      startRound();
    });
    sync();
  }

  // ---------- Loading flags ----------
  async function loadNextFlag() {
    for (let i = 0; i < MAX_LOAD_ATTEMPTS; i++) {
      const code = game.nextFlagCode();
      try {
        return { code, flag: await loadFlag(code) };
      } catch (err) {
        console.warn(`Skipping flag ${code}:`, err);
      }
    }
    throw new Error('Could not load any flag');
  }

  function prefetchIfNeeded() {
    const more = game.mode === 'infinite' || game.round < STANDARD_ROUNDS;
    if (!more) return;
    pending = loadNextFlag();
    pending.catch(() => {});
  }

  async function startRound() {
    app.innerHTML = `${gameHeader()}
      <section>
        <h1 class="title country">&nbsp;</h1>
        <div class="flag-frame loading"></div>
      </section>`;
    wireQuit();

    const promise = pending ?? loadNextFlag();
    pending = null;
    let next;
    try {
      next = await promise;
    } catch {
      renderLoadError();
      return;
    }
    if (!app.querySelector('.flag-frame.loading')) return; // user left meanwhile
    renderRound(next);
    prefetchIfNeeded();
  }

  function renderLoadError() {
    app.innerHTML = `${gameHeader()}
      <section>
        <p class="error">Couldn't load flags. Check your connection.</p>
        <button class="primary" id="retry">Retry</button>
      </section>`;
    wireQuit();
    app.querySelector('#retry').addEventListener('click', startRound);
  }

  // ---------- Round ----------
  function renderRound({ code, flag }) {
    app.innerHTML = `${gameHeader()}
      <section>
        <h1 class="title country"></h1>
        <div class="flag-frame"></div>
        <div class="controls">
          <div class="sliders">
            <input type="range" class="slider" id="hue" min="0" max="360" step="1" aria-label="Hue">
            <input type="range" class="slider" id="sat" min="0" max="100" step="1" aria-label="Saturation">
            <input type="range" class="slider" id="bri" min="0" max="100" step="1" aria-label="Brightness">
          </div>
          <button class="primary" id="lock">Lock Color ${CHECK_ICON}</button>
        </div>
      </section>`;
    wireQuit();
    app.querySelector('.country').textContent = flagByCode(code).name;
    flag.svg.setAttribute('aria-label', `Flag of ${flagByCode(code).name}`);
    app.querySelector('.flag-frame').append(flag.svg);

    const hue = app.querySelector('#hue');
    const sat = app.querySelector('#sat');
    const bri = app.querySelector('#bri');
    hue.value = Math.floor(Math.random() * 361);
    sat.value = 50;
    bri.value = 50;
    hue.style.background = HUE_GRADIENT;

    const current = () => hsbToHex(Number(hue.value), sat.value / 100, bri.value / 100);
    const paint = () => {
      const h = Number(hue.value), s = sat.value / 100, b = bri.value / 100;
      sat.style.background = `linear-gradient(to right, ${hsbToHex(h, 0, b)}, ${hsbToHex(h, 1, b)})`;
      bri.style.background = `linear-gradient(to right, #000000, ${hsbToHex(h, s, 1)})`;
      flag.setGuess(current());
    };
    for (const input of [hue, sat, bri]) input.addEventListener('input', paint);
    paint();

    app.querySelector('#lock').addEventListener('click', () => {
      const result = game.submit(code, current(), flag.hiddenHex);
      flag.reveal();
      renderReveal(result);
    });
  }

  function renderReveal(result) {
    if (game.mode === 'infinite') {
      // Lives and the streak changed. (Standard keeps showing the current round.)
      app.querySelector('.header').outerHTML = gameHeader();
      wireQuit();
    }
    const controls = app.querySelector('.controls');
    controls.innerHTML = `<div class="reveal">
        <div class="swatches">
          <div class="swatch"><div class="chip" id="chip-guess"></div>Yours<code id="hex-guess"></code></div>
          <div class="score"><strong id="score"></strong><span>/ 10</span></div>
          <div class="swatch"><div class="chip" id="chip-true"></div>Actual<code id="hex-true"></code></div>
        </div>
        <p class="verdict" id="verdict"></p>
        <button class="primary" id="next"></button>
      </div>`;
    controls.querySelector('#chip-guess').style.background = result.guessHex;
    controls.querySelector('#chip-true').style.background = result.trueHex;
    controls.querySelector('#hex-guess').textContent = result.guessHex;
    controls.querySelector('#hex-true').textContent = result.trueHex;
    controls.querySelector('#score').textContent = result.score.toFixed(1);
    const verdict = controls.querySelector('#verdict');
    verdict.textContent = verdictFor(result.score);
    if (result.lostLife) {
      const lost = document.createElement('span');
      lost.className = 'lost';
      lost.textContent = ' · −1 life';
      verdict.append(lost);
    }
    const next = controls.querySelector('#next');
    next.textContent = game.isOver() ? 'See results' : 'Next flag';
    next.addEventListener('click', () => (game.isOver() ? renderEnd() : startRound()));
    next.focus();
  }

  // ---------- End ----------
  function renderEnd() {
    pending = null;
    const standard = game.mode === 'standard';
    const best = game.bestFor(game.mode, game.tier);
    app.innerHTML = `${header({ sub: `${MODE_LABELS[game.mode]} · ${TIER_LABELS[game.tier]}` })}
      <section>
        <h1 class="title">${standard ? 'Results' : 'Out of lives'}</h1>
        <p class="big-score">${standard ? `${game.total}<span> / 50</span>` : `${game.results.length}<span> flags</span>`}</p>
        ${game.isNewBest ? '<span class="badge">New best!</span>' : `<p class="best">${formatBest(game.mode, best)}</p>`}
        <ol class="results"></ol>
        <button class="primary" id="again">Play again</button>
        <button class="secondary" id="home">Home</button>
      </section>`;
    const list = app.querySelector('.results');
    for (const r of game.results) {
      const li = document.createElement('li');
      li.innerHTML = `<span class="pair"><i></i><i></i></span><span class="name"></span><span class="pts"></span>`;
      const [guess, truth] = li.querySelectorAll('i');
      guess.style.background = r.guessHex;
      guess.title = `Yours ${r.guessHex}`;
      truth.style.background = r.trueHex;
      truth.title = `Actual ${r.trueHex}`;
      li.querySelector('.name').textContent = r.name;
      li.querySelector('.pts').textContent = r.score.toFixed(1);
      list.append(li);
    }
    app.querySelector('#again').addEventListener('click', () => {
      game.start(game.mode, game.tier);
      startRound();
    });
    app.querySelector('#home').addEventListener('click', renderHome);
  }

  renderHome();
}
```
